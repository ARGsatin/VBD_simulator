Load config.yaml:
  # --- 1. 物理环境与材料本构 (Hyperelastic Material) ---

[模块 1] 几何管线与共形预处理 (Geometry Pipeline)
=========================================================================================
1.1 共形网格生成 :
    # 工业级软件标准做法：切片不是独立切的，而是把层与层的交界面作为限制条件喂给网格划分器 (TetGen)。
    # 目的：保证第 k 层和第 k-1 层之间完全共享节点（共形拓扑），撕拉时才不会产生两张皮分离的错误。

1.2 空间分层与着色:
    # 给每个节点打上“出生层号”标签。
    # 图着色 (Graph Coloring) 是 GPU 并行计算的灵魂：相邻的节点不能是同一种颜色。
    # 这样在计算时，同色节点可以被成千上万个 GPU 线程同时处理，绝对不会发生两个线程抢着修改同一个四面体数据的冲突 (Race Condition)。


def execute_geometry_pipeline(stl_filepath, params):
    print(">>> 开始执行几何管线与预处理...")
    
# --------------------------------------------------------------------------
# 1.1 基于内部约束面的共形网格生成 (Conformal Meshing via PLC)
# --------------------------------------------------------------------------
 # 【物理意义】DLP 是逐层打印的。如果直接把整个模型划分网格，层与层的交界处
 # 会横穿四面体内部，导致我们无法“按层”激活物理属性。
# 必须强制网格生成器在 Z = k * layer_thickness 的地方生成绝对平齐的切面。
    
    # a. 读取外壳 STL
    bounding_box = compute_bounding_box(stl_filepath)
    Z_max = bounding_box.z_max
    total_layers = ceil(Z_max / params.layer_thickness)
    
    # b. 构建分层约束面 (Piecewise Linear Complex, PLC)
    internal_surfaces = []
    for k in range(1, total_layers):
        z_plane = k * params.layer_thickness
        # 计算平面与 STL 外壳的交线（切片轮廓多边形）
        slice_polygon = compute_intersection_polygon(stl_filepath, z_plane)
        internal_surfaces.append(slice_polygon)
        
    # c. 调用网格划分内核 (如 Gmsh 或 TetGen)
    mesher_config = {
        "preserve_boundaries": True, 
        "internal_facets": internal_surfaces, # 强制注入层间约束面
        "max_tet_volume": params.max_element_size 
    }
    raw_nodes, raw_tets = generate_tetrahedral_mesh(stl_filepath, mesher_config)
    
    # --------------------------------------------------------------------------
    # 1.2 空间映射、分层与生前状态绑定 (Spatial Sorting & Layer Assignment)
    # --------------------------------------------------------------------------
  # 【物理意义】确立每一个顶点和四面体的“出生层号”，并标记它们的边界属性。
    # 共享节点（处于分层面上）必须归属于它最早被打印出来的那一层。
    
    Nodes = []
    Tets = []
    
    # a. 四面体归属划分
    for i, tet_indices in enumerate(raw_tets):
        tet = Tetrahedron(id=i, node_indices=tet_indices)
        # 取四面体重心的 Z 坐标来判断其属于哪一层
        centroid_z = average([raw_nodes[idx].z for idx in tet_indices])
        tet.layer_id = floor(centroid_z / params.layer_thickness)
        Tets.append(tet)
        
    # b. 顶点全局唯一化与属性判定
    for i, coord in enumerate(raw_nodes):
        node = Node(id=i, x_ideal=coord)
        
        # 解决浮点数精度问题，寻找该节点所属的精确 Z 几何平面
        layer_float = coord.z / params.layer_thickness
        nearest_layer = round(layer_float)
        
        if abs(layer_float - nearest_layer) < 1e-6:
            # 该节点正好位于分层交界面上 (它是共享节点！)
            if nearest_layer == 0:
                node.is_bottom_surface = True
                node.first_active_layer = 0
            else:
                node.is_bottom_surface = False
                # 处于 k 层和 k-1 层交界的节点，随 k-1 层一起被“首次打印”出来
                node.first_active_layer = nearest_layer - 1 
                # 同时它也是 k 层的顶部节点
                node.is_top_surface_of_layer = nearest_layer 
        else:
            # 该节点悬浮在某一层内部
            node.is_bottom_surface = False
            node.first_active_layer = floor(layer_float)
            
        Nodes.append(node)

    # --------------------------------------------------------------------------
    # 1.3 拓扑图构建与 GPU 并行着色 (Graph Topology & Greedy Coloring)
    # --------------------------------------------------------------------------
    # 【算法意义】VBD 算法是"基于顶点的更新"。如果两个相邻顶点同时被不同的 GPU
    # 线程更新，它们会同时试图修改连接它们的那根"弹簧"（四面体）的数据，导致
    # 严重的显存竞态崩溃 (Race Condition)。
    # 必须用图着色算法：保证相连的节点颜色不同。按颜色批次推入 GPU 计算。
    
    # a. 构建邻接拓扑
    node_to_tets = build_inverse_mapping(Nodes, Tets)
    for node in Nodes:
        node.adjacent_tets = node_to_tets[node.id]
        # 计算节点分摊质量 (1/4 四面体质量累加)
        node.mass = 0.0
        for tet in node.adjacent_tets:
            node.mass += 0.25 * tet.volume * params.rho
            
        # 寻找拓扑相邻的节点 (只要共用任何一个四面体，即为相邻)
        node.neighbors = set()
        for tet in node.adjacent_tets:
            node.neighbors.update(tet.node_indices)
        node.neighbors.remove(node.id) # 移除自己
        
    # b. 贪心图着色算法 (Greedy Graph Coloring)
    Colors = [] # 存储不同颜色节点集合的列表
    
    # 为了着色质量更高（颜色总数更少），先按节点的度（邻居数量）降序排序
    sorted_nodes = sort_by_degree_descending(Nodes)
    
    for node in sorted_nodes:
        # 收集所有邻居已经占用的颜色
        used_colors = set(Nodes[nbr_id].color_id for nbr_id in node.neighbors if Nodes[nbr_id].color_id is not None)
        
        # 找到最小的未被使用的非负整数颜色
        assign_color = 0
        while assign_color in used_colors:
            assign_color += 1
            
        node.color_id = assign_color
        
        # 归档到颜色批次中
        while len(Colors) <= assign_color:
            Colors.append(ColorBatch(id=len(Colors)))
        Colors[assign_color].add_node(node.id)

    print(f">>> 预处理完成. 总节点数: {len(Nodes)}, 总四面体: {len(Tets)}.")
    print(f">>> GPU 着色完成. 共分配 {len(Colors)} 种颜色批次，可实现无锁高并发。")
    
    return Mesh(Nodes, Tets, Colors), total_layers

[模块 2] 逐层动态拉拔仿真闭环 (The Layer-by-Layer Loop)

# 初始化宏观数字孪生环境
Initialize: x = x_ideal, v = 0, E_z = 0, PID_integral = 0, prev_error = 0

For k = 0 to N-1 (开始逐层打印):

    -------------------------------------------------------------------------------------
    2.1 状态继承、硬约束碰撞与残余应力埋设 (最体现物理真实的阶段)
    -------------------------------------------------------------------------------------
    【步骤 A：平台下降与 FEP 膜硬碰撞检测】
    If k > 0:
        For idx in layer(k-1).bottom_nodes:
            If x[idx].z < Z_FEP_FILM:
                # 【物理现象】平台压得太低，上一层原本下垂的“肚子”撞到了树脂槽底部的玻璃。
                # 【数值处理】强制位移约束 (Kinematic Hard Constraint)，将其生硬地拍扁在 Z=0 处。
                x[idx].z = Z_FEP_FILM  
                v[idx].z = 0.0 # 撞墙后失去速度

    【步骤 B：新网格激活与防穿透插值】
    For idx in layer(k).nodes:
        If 属于顶部边界 (即共享节点):
            # 【几何遗传】新一层顶部被迫长成上一层底部的形状。如果上面下垂了，新层也跟着下垂。
            x[idx] = x_above[idx]      
        Else If 属于底部边界:
            x[idx].z = Z_FEP_FILM      # 底面被 FEP 膜绝对压平
        Else (属于夹在中间的新节点):
            # 【防爆处理】按实际上下边界的距离做线性插值。绝不能用CAD坐标，否则会刺穿上一层的网格，导致体积变负，求解器崩溃！
            x[idx] = lerp(x_above_boundary[idx], Z_FEP_FILM)
        v[idx] = 0

    【步骤 C：基矩阵预计算 (埋设收缩预应力，极度优雅的设计)】
    For tet in layer(k).tets:
        # 【物理黑魔法】我们把网格缩小 2% (c_shrink) 后的形状，告诉计算机：“这就是你没有受力时最舒服的形状”。
        # 结果：计算机一算当前坐标，发现“哎呀，我怎么比最舒服的形状大了一圈？我被拉伸了！”
        # 于是，网格自发产生了向内收缩的拉应力。这完美模拟了交联反应带来的体积收缩翘曲！
        tet.Dm = Compute_Edge_Matrix(tet.nodes.x * c_shrink) 
        tet.Dm_inv = Inverse(tet.Dm)  # 求逆后存入显存，后续求形变梯度 F 时直接矩阵乘法，快如闪电。
    
    【步骤 D：CZM 状态机初始化】
    Bottom_Nodes = layer(k).bottom_nodes
    For i in Bottom_Nodes:
        i.czm_state = "FIXED"       # 刚固化好，死死黏在膜上
        i.damage = 0.0              # 一点损伤都没有
        i.time_free = 0.0           # 自由时间归零
    Top_Nodes.is_fixed = True       # 顶部被成型台死死拽住

    -------------------------------------------------------------------------------------
    2.2 时间步进与多物理场 VBD 求解 (最吃算力的微分方程求解区)
    -------------------------------------------------------------------------------------
    t = 0.0; stable_counter = 0; is_lifting = True
    
    While True (在时间长河中不断推进，直到彻底安静):
        
        [2.2.1 平台运动学驱动 (Kinematics)]
        If is_lifting:
            For i in Top_Nodes: 
                x[i].z += v_lift * dt   # 成型台匀速向上拉扯，给系统持续注入弹性势能
                v[i].z = v_lift         # 赋予连续速度，这是计算流体和阻尼的基石
            
        [2.2.2 界面相变判定 (CZM State Machine)]
        For i in Bottom_Nodes:
            # 状态 1：完全黏着
            If i.czm_state == "FIXED":
                F_pull = get_internal_force_z(i) # 计算水凝胶内部弹簧把它往上拽的力
                If (F_pull / A_i) >= T_max:
                    # 拉力超过极限，胶水撑不住了，进入撕裂状态！
                    i.czm_state = "DAMAGING"  
            
            # 状态 2：撕裂损伤区 (平滑过渡防数值爆炸)
            Else If i.czm_state == "DAMAGING":
                gap = x[i].z - Z_FEP_FILM
                # 线性软化定律：拉得越开，损伤度 D 越大，残余黏性越小。
                D = max(i.damage, (gap / delta_f) * (T_max / K_czm))
                i.damage = clamp(D, 0.0, 1.0)
                If gap >= delta_f or i.damage >= 1.0:
                    i.czm_state = "FREE"      # 胶水彻底断开，物理上实现完全脱膜！
                
        [2.2.3 VBD 非线性求解与加速引擎 (Vertex Block Descent的核心)]
        x_prev = copy(x) # 【双缓冲技术】拍下上一帧的照片，用于计算当前帧的速度。
        
        【加速引擎 1：自适应初始化】
        For i in Free_Nodes: 
            # 如果直接用重力 g 初始化，系统一开始会猛地往下掉。用自适应加速度能平滑启动。
            a_ext = estimate_adaptive_accel(f_internal_prev, m_i, c_init)
            y_i = x_prev_i + dt * v_i + (dt^2) * a_ext 
            x[i] = y_i # 得到本轮迭代的“初始猜测位置”
                
        # --- 牛顿-拉夫逊 (Newton-Raphson) 局部求根过程 ---
        For iter = 1 to max_iters:
            x_old_iter = copy(x) 
            max_dx = 0.0 # 记录这一圈所有节点里挪动最大的距离，用于判断是否收敛
            
            For each color in Colors:
                # --- 【GPU 极速并行区】同颜色的节点互不影响，一并计算 ---
                For each vertex i in color:
                    If i.is_top_fixed or i.czm_state == "FIXED": 
                        Continue # 【绝对高效】既然被固定死了，算也没用，直接跳过，省下大量矩阵求逆的算力。
                    
                    # 1. 结构力学系统组装
                    f_inertia = -(m_i / dt^2) * (x_i - y_i) # 惯性力：阻止物体改变当前运动状态
                    f_elastic, H_elastic = compute_hyperelasticity(i, tet.Dm_inv) # 根据 F=Ds*Dm_inv 算超弹性力
                    f_damp    = -(k_d / dt) * H_elastic * (x_i - x_prev_i) # 瑞利阻尼：消耗动能，让震荡停下来
                    H_inertia = (m_i / dt^2) * I
                    H_damp    = (k_d / dt) * H_elastic
                    
                    # 2. 内聚力平滑过渡 (CZM 软化)
                    f_czm = [0,0,0]; H_czm = Zero_3x3()
                    If i.czm_state == "DAMAGING":
                        # 提供一个逐渐衰减的向下拉力，防止瞬间断开导致受力台阶突变（数值震荡）
                        t_curr = (1.0 - i.damage) * T_max * (1.0 - max(x_i.z, 0)/delta_f)
                        f_czm.z = - t_curr * A_i
                        H_czm[2,2] = - ((1.0 - i.damage) * T_max / delta_f) * A_i # 负刚度矩阵项
                    
                    # 3. 极值流体负压 (Stefan 抽吸流)
                    f_fluid = [0,0,0]; H_fluid = Zero_3x3()
                    If i.czm_state == "FREE":
                        gap = max(x_i.z - Z_FEP_FILM, d_min) # 【防爆】防止紧贴时 gap=0 导致除以 0 报错。
                        If gap < d_fluid_max and i.time_free < t_fluid_max:
                            # 间隙极小时，v_z_imp（向上抬的速度）会引发极其恐怖的流体负压（吸力）
                            v_z_imp = (x_i.z - x_prev_i.z) / dt
                            f_fluid.z = -(C_0 * eta * R^4 / gap^3) * v_z_imp
                            H_fluid[2,2] = (C_0 * eta * R^4 / (dt * gap^3)) # 隐式流体刚度，保证算法稳定性
                            
                    # 4. 电场托举力 (核心干预手段)
                    f_elec = [0, 0, q_ion * E_z] # Q*E = F，极其简洁的体积力
                    
                    # 5. 系统总装与求解 (H * dx = f)
                    f_total = f_inertia + f_elastic + f_damp + f_czm + f_fluid + f_elec # 总残差力
                    H_total = H_inertia + H_elastic + H_damp + H_czm + H_fluid # 总刚度（海森矩阵）
                    
                    dx_i = Inverse(H_total) * f_total # 解 3x3 线性方程，算出这一步该走多远
                    x_i += dx_i # 挪动节点位置
                    max_dx = max(max_dx, length(dx_i))
            
            【加速引擎 2：切比雪夫外推 (Chebyshev Momentum)】
            If iter > 5: 
                omega = compute_chebyshev_omega(iter, rho_cheb)
                For i in Free_Nodes:
                    # 【黑魔法】顺着这几次挪动的趋势，胆子大一点，往后多推算一步，加速收敛！
                    # 但处于撕裂边缘 (DAMAGING) 的节点极不稳定，屏蔽外推防止算爆炸。
                    If i.czm_state != "DAMAGING": 
                        x[i] += omega * (x[i] - x_old_iter[i])
                        
            # 如果整张网格动得最大的人，也挪动不到一微米，说明局部受力平衡了，进入下一个物理时间。
            If max_dx < epsilon: Break 

        [2.2.4 状态流转与真实速度更新]
        For i in Free_Nodes: v[i] = (x[i] - x_prev[i]) / dt # 经过疯狂迭代稳定后的真实隐式速度
        For i in Bottom_Nodes: 
            If i.czm_state == "FREE": i.time_free += dt # 记录自由了多久，用来干掉流体负压
        t += dt
        
        [2.2.5 严格动能静平衡判定]
        If Check_All_Free(Bottom_Nodes) and is_lifting:
            is_lifting = False # 全部脱膜，电机停机，不再拉扯
            
        If not is_lifting:
            E_k = compute_total_kinetic_energy()
            # 必须满足动能极小（阻尼耗尽）且没有任何节点瞎乱窜
            If E_k < 1e-6 and max(v) < 1e-4:
                stable_counter += 1
            Else:
                stable_counter = 0
                
            If stable_counter >= N_stable:
                Break # 连续 10 步保持死寂，说明悬垂结构彻底挂稳了，退出微观时间循环！

    -------------------------------------------------------------------------------------
    2.3 形态量化评估 (数字孪生数据采集器)
    -------------------------------------------------------------------------------------
    # 【过滤机制】不采单个最值（防止数值奇点），取底层所有节点的平均下垂误差
    sum_z_error = 0.0
    For idx in Bottom_Nodes:
        sum_z_error += (x_ideal[idx].z - x[idx].z) # CAD 理想尺子 - 残酷物理现实
    err_avg = sum_z_error / count(Bottom_Nodes)

    -------------------------------------------------------------------------------------
    2.4 PID 闭环电场控制 (数字孪生的智慧大脑)
    -------------------------------------------------------------------------------------
    If err_avg > err_target:
        # 如果下垂超标，根据误差 (e_n) 猛下猛药，计算下一层所需的电场补偿增量
        e_n = err_avg - err_target
        PID_integral += e_n * dt
        e_derivative = (e_n - prev_error) / dt
        
        delta_E = K_p * e_n + K_i * PID_integral + K_d * e_derivative
        # 【电气安全】加上算好的增量，并使用 clamp 限制在 0-500V 内，防止积分饱和把电路烧穿
        E_z = clamp(E_z + delta_E, 0, E_max)  
        
        prev_error = e_n

    -------------------------------------------------------------------------------------
    2.5 制造协议下发 (从虚拟走向现实)
    -------------------------------------------------------------------------------------
    Append_To_Gcode("M150 E{E_z}") # 将刚算好的电场电压，无缝插进真实打印机的控制代码中
    Export_VTU(layer=k, mesh=x, stress=compute_von_mises()) # 倒出渲染用的彩色三维模型数据
